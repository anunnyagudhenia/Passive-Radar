CM 7-el Yagi @255 MHz (compromise: keep gain high, SWR < 2)
CM Forw: 90,0 ; Back: 90,180
CE
' Elements along Z; boom along X; units = meters

SY rw=0.01       ' element radius (20 mm dia)
SY seg=41
SY freq=255

' Element lengths (meters) - compromise values
SY Lr=0.590      ' reflector length (slightly longer than driven)
SY Ld=0.556      ' driven length
SY L1=0.528      ' director1
SY L2=0.518
SY L3=0.506
SY L4=0.499
SY L5=0.493

' Half-lengths
SY hr=Lr/2
SY hd=Ld/2
SY h1=L1/2
SY h2=L2/2
SY h3=L3/2
SY h4=L4/2
SY h5=L5/2

' X positions (absolute, meters) - moderate boom
SY xr=-0.195     ' reflector (a little closer than high-gain layout)
SY xd=0.0
SY x1=0.180
SY x2=0.390
SY x3=0.680
SY x4=1.030
SY x5=1.390

' ------- GEOMETRY --------
GW 1 41  xr 0 -hr   xr 0 hr  rw    ' Reflector
GW 2 41  xd 0 -hd   xd 0 hd  rw    ' Driven (feed at center)
GW 3 41  x1 0 -h1   x1 0 h1  rw    ' Director 1
GW 4 41  x2 0 -h2   x2 0 h2  rw    ' Director 2
GW 5 41  x3 0 -h3   x3 0 h3  rw    ' Director 3
GW 6 41  x4 0 -h4   x4 0 h4  rw    ' Director 4
GW 7 41  x5 0 -h5   x5 0 h5  rw    ' Director 5

GE 0
GN -1

EX 0 2 21 0 1 0     ' feed at center of driven (seg=41 -> center=21)
FR,0,1,0,0,255
PQ -1
PT -1
RP,0,361,1,1000,0,0,1,0,90   ' azimuth sweep, theta=90 (boresight)

EN
